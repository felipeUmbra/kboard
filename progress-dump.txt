// Auto-progress calculation for Epics and Stories.

import { useMemo } from "react";
import type { Board, Card, CardType } from "./types";

export interface CardProgress {
  total: number;
  done: number;
  /** null when there are no children (so we show "—" instead of "0%"). */
  percent: number | null;
}

/** A lookup that maps a card id to the id of the column it currently lives in. */
function buildColumnIndex(board: Board): Map<string, string> {
  const idx = new Map<string, string>();
  for (const col of board.columns) {
    for (const cardId of col.cardIds) {
      idx.set(cardId, col.id);
    }
  }
  return idx;
}

/**
 * Compute progress for a card based on its descendants.
 * - Tasks return `{total:0, done:0, percent:null}` (no children)
 * - Stories count direct child tasks
 * - Epics count direct child stories + tasks of those stories (transitive tasks)
 */
export function computeProgress(
  card: Card,
  board: Board,
): CardProgress {
  const doneSet = new Set(board.doneColumnIds);
  const colIdx = buildColumnIndex(board);
  const cards = board.cards;

  const isDone = (id: string): boolean => {
    const colId = colIdx.get(id);
    return colId !== undefined && doneSet.has(colId);
  };

  const collectChildTasks = (parentId: string): Card[] => {
    const out: Card[] = [];
    for (const c of Object.values(cards)) {
      if (c.type === "task" && c.parentIds.includes(parentId)) out.push(c);
    }
    return out;
  };
  const collectChildStories = (parentId: string): Card[] => {
    const out: Card[] = [];
    for (const c of Object.values(cards)) {
      if (c.type === "story" && c.parentIds.includes(parentId)) out.push(c);
    }
    return out;
  };

  let total = 0;
  let done = 0;

  if (card.type === "task") {
    return { total: 0, done: 0, percent: null };
  }
  if (card.type === "story") {
    const children = collectChildTasks(card.id);
    total = children.length;
    done = children.filter((c) => isDone(c.id)).length;
  } else if (card.type === "epic") {
    // Count stories directly parented + tasks of those stories.
    const stories = collectChildStories(card.id);
    const storyIds = new Set(stories.map((s) => s.id));
    const tasks: Card[] = [];
    for (const c of Object.values(cards)) {
      if (c.type !== "task") continue;
      if (c.parentIds.some((p) => storyIds.has(p))) tasks.push(c);
    }
    const all = [...stories, ...tasks];
    total = all.length;
    done = all.filter((c) => isDone(c.id)).length;
  }

  return {
    total,
    done,
    percent: total === 0 ? null : Math.round((done / total) * 100),
  };
}

/** React hook with memoization. Keyed on a coarse hash of board mutations. */
export function useProgress(card: Card, board: Board): CardProgress {
  const hash = useMemo(
    () =>
      board.cards[card.id]?.parentIds?.join(",") ?? "" +
      "|" +
      board.doneColumnIds.join(",") +
      "|" +
      board.columns.map((c) => c.id + ":" + c.cardIds.length).join(","),
    [board, card.id, card.parentIds],
  );
  return useMemo(() => computeProgress(card, board), [card, board, hash]);
}

/** Count cards of a given type on the board. */
export function countByType(
  board: Board,
  type: CardType,
): number {
  let n = 0;
  for (const c of Object.values(board.cards)) {
    if (c.type === type) n++;
  }
  return n;
}
